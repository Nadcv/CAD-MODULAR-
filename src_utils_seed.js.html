import { db } from '../firebase'; import { collection, doc, setDoc, getDocs, query, limit } from 'firebase/firestore'; import { questions_es } from '../data/questions_es'; import { questions_pt } from '../data/questions_pt'; export async function seedQuestions() { const qRef = collection(db, 'questions'); const snap = await getDocs(query(qRef, limit(1))); if (!snap.empty) { console.log('✅ Perguntas já existem no Firestore.'); return; } // Combinar ES e PT no mesmo documento for (let i = 0; i < questions_es.length; i++) { const esQ = questions_es[i]; const ptQ = questions_pt.find(q => q.id === esQ.id); await setDoc(doc(db, 'questions', esQ.id), { id: esQ.id, tema: esQ.tema, texto_es: esQ.texto, texto_pt: ptQ ? ptQ.texto : esQ.texto, profundidad: esQ.profundidad }); console.log(`✅ Seed: ${esQ.id}`); } console.log('🚀 Todas as perguntas foram carregadas!'); }
